import { cleanup, fireEvent, render, screen, waitFor } from "@testing-library/react";
import { afterEach, beforeEach, expect, it, vi } from "vitest";

const mocks = vi.hoisted(() => ({
  settings: { disableAudio: false },
  doPermissionsCheck: vi.fn(),
  requestPermission: vi.fn(),
  requestPermissionWithFlow: vi.fn(),
}));
vi.mock("@/lib/hooks/use-settings", () => ({ useSettings: () => ({ settings: mocks.settings, isSettingsLoaded: true }) }));
vi.mock("@/lib/hooks/use-platform", () => ({ usePlatform: () => ({ isMac: true }) }));
vi.mock("@/lib/hooks/use-interval", () => ({ useInterval: vi.fn() }));
vi.mock("@/lib/hooks/use-tauri-event", () => ({ useTauriEvent: vi.fn() }));
vi.mock("@/lib/utils/tauri", () => ({ commands: mocks }));
vi.mock("@/lib/utils/permission-flow", () => ({ requestPermissionWithFlow: mocks.requestPermissionWithFlow, openPermissionSettingsWithFlow: vi.fn() }));

import { PermissionBanner } from "./permission-banner";

beforeEach(() => {
  vi.clearAllMocks();
  mocks.settings.disableAudio = false;
  mocks.doPermissionsCheck.mockResolvedValue({ microphone: "denied", screenRecording: "granted", accessibility: "granted" });
});

it("removes a microphone warning immediately when audio is disabled", async () => {
  const view = render(<PermissionBanner />);
  expect(await screen.findByText("microphone disabled")).toBeInTheDocument();
  mocks.settings.disableAudio = true;
  view.rerender(<PermissionBanner />);
  expect(screen.queryByRole("button", { name: "Fix permissions" })).toBeNull();
  expect(mocks.requestPermission).not.toHaveBeenCalled();
});

it("still recovers screen access when audio is disabled and microphone is denied", async () => {
  mocks.settings.disableAudio = true;
  mocks.doPermissionsCheck.mockResolvedValue({ microphone: "denied", screenRecording: "denied", accessibility: "granted" });
  render(<PermissionBanner />);
  fireEvent.click(await screen.findByRole("button", { name: "Fix permissions" }));
  await waitFor(() => expect(mocks.requestPermissionWithFlow).toHaveBeenCalledWith("screenRecording"));
  expect(mocks.requestPermission).not.toHaveBeenCalled();
});

it("requests microphone permission when audio is enabled", async () => {
  render(<PermissionBanner />);
  fireEvent.click(await screen.findByRole("button", { name: "Fix permissions" }));
  await waitFor(() => expect(mocks.requestPermission).toHaveBeenCalledWith("microphone"));
});

afterEach(() => { cleanup(); });
