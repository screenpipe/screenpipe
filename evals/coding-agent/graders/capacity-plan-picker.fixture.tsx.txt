const blockedFetch = vi.fn(async () => { throw new Error("Unexpected network request"); });
function user(plan = "pro", token: string | null = "synthetic-token") {
  mocks.state.user = { id: "synthetic-user", email: "member@example.test", token,
    cloud_subscribed: true, subscription_plan: plan };
}
// Find the smallest visible grouping containing the plan name and price. No
// class, test id, internal card key, helper name or DOM depth is prescribed.
function card(name: string, price: number): HTMLElement {
  const label = screen.getByText(new RegExp(`^${name}$`, "i"));
  let el = label.parentElement;
  while (el && !within(el).queryByText(`$${price}`)) el = el.parentElement;
  expect(el, `visible ${name} price grouping`).not.toBeNull();
  return el!;
}
function billing(target: string) {
  expect(mocks.openUrl).toHaveBeenCalledTimes(1);
  const url = new URL(mocks.openUrl.mock.calls[0][0]);
  expect(url.origin).toBe("https://screenpipe.com");
  expect(url.pathname).toBe("/account/billing");
  expect(url.searchParams.get("target_plan")).toBe(target);
  expect(url.searchParams.get("interval")).toBe("month");
  expect(blockedFetch).not.toHaveBeenCalled();
}

describe("capacity plan outcomes", () => {
  it("exposes all five named prices without a feature flag", () => {
    mount();
    for (const [name, price] of [["free",0],["basic",25],["business",50],["business max",100],["business ultra",200]] as const)
      expect(card(name,price)).toBeVisible();
    expect(card("business max",100).textContent).toMatch(/2x.*quer/i);
    expect(card("business ultra",200).textContent).toMatch(/4x.*quer/i);
    for (const [name,price] of [["business max",100],["business ultra",200]] as const) {
      expect(card(name,price).textContent).toMatch(/400.*credits/i);
      expect(card(name,price).textContent).toMatch(/frontier.*models/i);
    }
  });
  it("selects Max through website billing without starting checkout", () => {
    mount(); fireEvent.click(within(card("business max",100)).getByRole("button")); billing("pro_max");
  });
  it("selects Ultra from Max through website billing", () => {
    mount("pro_max"); fireEvent.click(within(card("business ultra",200)).getByRole("button")); billing("pro_ultra");
  });
  it.each([["pro_max","business max",100],["pro_ultra","business ultra",200]] as const)("marks the actual %s plan current", (plan,name,price) => {
    mount(plan); const active=card(name,price);
    expect(within(active).getAllByText(/^(current|your plan)$/i).length).toBeGreaterThan(0);
    expect(within(card("business",50)).queryByText(/^(current|your plan)$/i)).toBeNull();
    expect(within(active).queryByRole("button")).toBeNull();
  });
  it("recommends the immediate capacity step and none above or off ladder", () => {
    const view=mount();
    expect(within(card("business max",100)).getByText(/next step/i)).toBeVisible();
    user("pro_max");view.rerender(<AccountSection />);
    expect(within(card("business ultra",200)).getByText(/next step/i)).toBeVisible();
    for (const plan of ["pro_ultra","team","enterprise"]) {
      user(plan);view.rerender(<AccountSection />);expect(screen.queryByText(/next step/i)).toBeNull();
    }
  });
  it("preserves signed-out stale-account login gating", () => {
    vi.stubGlobal("fetch",blockedFetch);user("pro",null);render(<AccountSection />);
    expect(screen.getByText(/sign in to screenpipe/i)).toBeVisible();
    expect(screen.getByText(/not logged in/i)).toBeVisible();
    expect(screen.queryByText(/^active$/i)).toBeNull();
  });
  it("preserves signed-in account access and the account link", () => {
    mount();expect(screen.getByText(/^active$/i)).toBeVisible();
    expect(screen.getByText(/logged in as member@example.test/i)).toBeVisible();
    fireEvent.click(screen.getByRole("button",{name:/web account/i}));
    expect(mocks.openUrl).toHaveBeenCalledWith("https://screenpipe.com/account");
  });
});
