import {beforeEach,afterEach,expect,mock,test} from 'bun:test';
let state:any;
mock.module('@sentry/cloudflare',()=>({captureException:(error:any)=>state.alerts.push(error.message)}));
mock.module('../services/model-health',()=>({logModelOutcome:async(_env:any,entry:any)=>state.health.push(entry)}));
mock.module('../providers',()=>({resolveModelAlias:(model:string)=>model,createProvider:(model:string)=>({supportsTools:true,createCompletion:async(body:any)=>deliver(model,body,false),createStreamingCompletion:async(body:any)=>(await deliver(model,body,true)).body})}));
const {handleChatCompletions}=await import('../handlers/chat');
const originalFetch=globalThis.fetch;
const cap='400 {"error":{"type":"invalid_request_error","message":"You have reached your specified API usage limits. You will regain access on 2030-01-01 at 00:00 UTC."},"request_id":"synthetic-provider-id"}';
const payload={id:'synthetic-reply',choices:[{message:{role:'assistant',content:'synthetic answer'}}]};
beforeEach(()=>{state={calls:[],health:[],alerts:[],network:[],mode:'cap',message:cap,status:400};globalThis.fetch=(async(...args:any[])=>{state.network.push(String(args[0]));throw Error('unexpected external request')}) as any});
afterEach(()=>{globalThis.fetch=originalFetch;expect(state.network).toEqual([])});
async function deliver(model:string,body:any,stream:boolean){state.calls.push({model,body:structuredClone(body)});if(state.mode!=='success'&&!(state.mode==='recover'&&state.calls.length>1))throw Object.assign(Error(state.message),{status:state.status});return stream?new Response('data: '+JSON.stringify(payload)+'\n\ndata: [DONE]\n\n'):Response.json(payload)}
function request(extra:any={}){return {model:'synthetic-single-model',messages:[{role:'system',content:'screenpipe synthetic instruction'},{role:'user',content:'synthetic request'}],max_tokens:120,tools:[{type:'function',function:{name:'lookup',parameters:{type:'object',properties:{}}}}],stream:false,...extra}}
function invoke(body:any){return handleChatCompletions(body,{} as any,'interactive','',false,{backgroundFallback:false,safetyRefusalFallback:false} as any)}
function advisory(text:string){expect(text).toMatch(/provider/i);expect(text).toMatch(/capacity|usage limit/i);expect(text).toMatch(/try.*later|try again|different provider|reset/i);expect(text).not.toMatch(/invalid_request_error|synthetic-provider-id|malformed tool|unsupported parameter/i)}
for(const stream of [false,true])test('single model cap returns safe recovery guidance '+stream,async()=>{const response=await invoke(request({stream}));expect(response.status).toBe(400);advisory(await response.text());expect(state.calls.length).toBe(1);expect(state.alerts).toEqual([]);expect(state.health).toContainEqual({model:'synthetic-single-model',outcome:'error'})});
test('exhausted explicit fallback chain returns provider recovery guidance',async()=>{const response=await invoke(request({model:'claude-sonnet-5'}));expect(response.status).toBe(400);expect(state.calls.length).toBeGreaterThan(1);advisory(await response.text());expect(state.alerts).toEqual([])});
test('provider cap preserves successful fallback and original request',async()=>{state.mode='recover';const body=request({model:'claude-sonnet-5'}),saved=structuredClone(body);const response=await invoke(body);expect(response.status).toBe(200);expect(await response.json()).toEqual(payload);expect(state.calls.length).toBeGreaterThan(1);expect(state.calls[1].body.messages).toEqual(saved.messages);expect(state.calls[1].body.tools).toEqual(saved.tools);expect(body).toEqual(saved)});
test('healthy explicit provider remains successful',async()=>{state.mode='success';const response=await invoke(request());expect(response.status).toBe(200);expect(await response.json()).toEqual(payload);expect(state.calls.length).toBe(1)});
test('malformed image remains a client payload error',async()=>{state.message='Failed to decode image data';const response=await invoke(request());expect(response.status).toBe(400);const text=await response.text();expect(text).toMatch(/image/i);expect(text).not.toMatch(/provider.*usage limit|temporarily at capacity/i);expect(state.calls.length).toBe(1)});
test('empty request fails before provider work',async()=>{expect((await invoke(request({messages:[]}))).status).toBe(400);expect(state.calls).toEqual([])});
test('provider authentication failure retains unauthorized status',async()=>{state.status=401;state.message='invalid API key';const response=await invoke(request());expect(response.status).toBe(401);expect(await response.text()).toContain('invalid API key');expect(state.calls.length).toBe(1)});
test('ordinary throttle retains retry guidance without cap advisory',async()=>{state.status=429;state.message='Too many requests';const response=await invoke(request());expect(response.status).toBe(429);const text=await response.text();expect(text).toMatch(/rate limit|try again/i);expect(text).not.toMatch(/provider.*usage limit|temporarily at capacity/i);expect(state.calls.length).toBe(1)});
