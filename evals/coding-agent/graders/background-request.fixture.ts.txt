import {beforeEach,afterEach,expect,mock,test} from 'bun:test';
let state: any;
mock.module('@sentry/cloudflare',()=>({captureException:()=>{}}));
mock.module('../services/model-health',()=>({logModelOutcome:async()=>{}}));
mock.module('../providers',()=>({resolveModelAlias:(model:string)=>model,createProvider:(model:string,_env:unknown,connection:unknown)=>({supportsTools:true,createCompletion:async(body:any)=>deliver(model,body,connection,false),createStreamingCompletion:async(body:any)=>(await deliver(model,body,connection,true)).body})}));
const {HostedChatAllowanceExceededError}=await import('../services/cloudflare-ai-gateway');
const {handleChatCompletions}=await import('../handlers/chat');
const originalFetch=globalThis.fetch;
beforeEach(()=>{state={calls:[],success:false,rescueFails:false,error:null,network:[]};globalThis.fetch=(async(url:any)=>{state.network.push(String(url));throw Error('unexpected network')}) as any});
afterEach(()=>{globalThis.fetch=originalFetch;expect(state.network).toEqual([])});
const env={OPENAI_API_KEY:'synthetic-key',SCREENPIPE_QWEN35_API_KEY:'synthetic-key',SCREENPIPE_QWEN35_URL:'https://synthetic.invalid'} as any;
const context={user_id:'synthetic',plan:'business_max',lane:'explicit',workload:'background',trial:false} as any;
const payload={id:'reply_synthetic',choices:[{message:{role:'assistant',tool_calls:[{id:'call_synthetic',type:'function',function:{name:'lookup',arguments:'{"query":"synthetic"}'}}]},finish_reason:'tool_calls'}]};
async function deliver(model:string,body:any,connection:unknown,stream:boolean){state.calls.push({model,body:structuredClone(body),connection});if(model==='fixture-primary'&&!state.success)throw state.error??new HostedChatAllowanceExceededError(context);if(model!=='fixture-primary'&&state.rescueFails)throw Object.assign(new Error('synthetic rescue unavailable'),{status:503});return stream?new Response('data: '+JSON.stringify(payload)+'\n\ndata: [DONE]\n\n'):Response.json(payload)}
function body(extra:any={}){return {model:'fixture-primary',stream:false,messages:[{role:'system',content:'screenpipe synthetic context'},{role:'developer',content:'Retain these synthetic task requirements.'},{role:'user',content:'old-context-start:'+ 'sample '.repeat(4000)+':old-context-end'},{role:'assistant',content:null,tool_calls:[{id:'call_prior',type:'function',function:{name:'lookup',arguments:'{"query":"past"}'}}]},{role:'tool',tool_call_id:'call_prior',content:'Synthetic prior tool result'},{role:'user',content:'Finish the synthetic summary.'}],tools:[{type:'function',function:{name:'lookup',description:'Required synthetic tool semantics. '.repeat(20),parameters:{type:'object',title:'SyntheticQuery',description:'Preserve schema guidance',properties:{query:{type:'string',description:'Synthetic search terms',default:'synthetic',examples:['example']}},required:['query']}}}],tool_choice:'auto',response_format:{type:'json_object'},max_completion_tokens:32000,temperature:0.2,...extra}}
async function invoke(request:any,enabled=true,settings:any=env){return handleChatCompletions(request,settings,'background','',false,{backgroundFallback:enabled,argusBackgroundFallback:enabled,safetyRefusalFallback:false,argusSafetyRefusalFallback:false} as any)}
for(const field of ['messages','tools','max_completion_tokens','response_format'])test('allowance rescue preserves '+field,async()=>{const request=body();const saved=structuredClone(request);const response=await invoke(request);expect(response.status).toBe(200);expect(state.calls.length).toBe(2);expect(state.calls[1].model).not.toBe('fixture-primary');expect(state.calls[1].body[field]).toEqual(saved[field]);expect(request).toEqual(saved);expect(await response.json()).toEqual(payload)});
test('streaming rescue preserves context and emits provider tool result',async()=>{const request=body({stream:true});const response=await invoke(request);expect(response.status).toBe(200);expect(state.calls.length).toBe(2);const {model,...actual}=state.calls[1].body;const {model:_requested,...expected}=request;expect(actual).toEqual(expected);expect(await response.text()).toContain('call_synthetic')});
test('successful primary needs no rescue',async()=>{state.success=true;expect((await invoke(body())).status).toBe(200);expect(state.calls.length).toBe(1)});
test('disabled rescue preserves allowance response',async()=>{const response=await invoke(body(),false);expect(response.status).toBe(429);expect(await response.text()).toContain('hosted_ai_allowance_exceeded');expect(state.calls.length).toBe(1)});
for(const type of ['image_url','file'])test('unsupported '+type+' remains refused without input loss',async()=>{const response=await invoke(body({messages:[{role:'user',content:[{type,[type]:{url:'https://synthetic.invalid/input'}}]}]}));expect(response.status).toBe(429);expect(state.calls.length).toBe(1)});
test('unconfigured rescue preserves allowance refusal',async()=>{expect((await invoke(body(),true,{})).status).toBe(429);expect(state.calls.length).toBe(1)});
test('unavailable rescue preserves original allowance error',async()=>{state.rescueFails=true;const response=await invoke(body());expect(response.status).toBe(429);expect(await response.text()).toContain('hosted_ai_allowance_exceeded');expect(state.calls.length).toBe(2)});
test('ordinary throttle cannot activate allowance rescue',async()=>{state.error=Object.assign(new Error('500 requests per minute'),{status:429});expect((await invoke(body())).status).toBe(429);expect(state.calls.length).toBe(1)});
test('empty messages are refused before provider work',async()=>{expect((await invoke(body({messages:[]}))).status).toBe(400);expect(state.calls).toEqual([])});
