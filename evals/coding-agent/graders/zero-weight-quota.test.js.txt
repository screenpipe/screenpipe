import { afterAll, expect, setSystemTime, test } from 'bun:test';
import { Database } from 'bun:sqlite';
import { trackUsage } from './usage-tracker';
setSystemTime(new Date('2026-01-15T12:00:00Z'));
afterAll(() => setSystemTime());

// Only the D1 transport is adapted: the real module issues SQL to SQLite.
for (const [name, count, model, ipCount, allowed] of [
  ['free above cap', 51, 'auto', null, true],
  ['free far above cap', 500, 'auto', null, true],
  ['free at cap', 50, 'auto', null, true],
  ['free below cap', 49, 'auto', null, true],
  ['paid at cap', 50, 'gpt-4o', null, false],
  ['anonymous IP limit', 51, 'auto', 1500, false],
  ['new free device', null, 'auto', null, true],
]) test(name, async () => {
  const db = new Database(':memory:');
  db.exec('CREATE TABLE usage (device_id TEXT PRIMARY KEY,user_id TEXT,daily_count INTEGER,last_reset TEXT,tier TEXT,updated_at TEXT)');
  const today = new Date().toISOString().slice(0, 10);
  if (count !== null) db.query('INSERT INTO usage VALUES(?,?,?,?,?,NULL)').run('synthetic-device', null, count, today, 'logged_in');
  if (ipCount !== null) db.query('INSERT INTO usage VALUES(?,?,?,?,?,NULL)').run('ip:192.0.2.1', null, ipCount, today, 'ip_tracking');
  function statement(sql, args = []) {
    return {
      bind: (...values) => statement(sql, values),
      first: async (column) => { const row = db.query(sql).get(...args); return column ? row?.[column] ?? null : row; },
      all: async () => ({ success: true, results: db.query(sql).all(...args), meta: {} }),
      run: async () => ({ success: true, results: [], meta: db.query(sql).run(...args) }),
    };
  }
  const env = { LIMIT_LOGGED_IN_DAILY: '50', DB: {
    prepare: (sql) => statement(sql),
    batch: async (items) => Promise.all(items.map(item => item.all())),
  } };
  const original = globalThis.fetch; let fetches = 0;
  globalThis.fetch = async () => { fetches++; throw new Error('Unexpected external call'); };
  try {
    const result = await trackUsage(env, 'synthetic-device', ipCount === null ? 'logged_in' : 'anonymous', undefined, ipCount === null ? undefined : '192.0.2.1', model);
    expect(result.allowed).toBe(allowed);
    expect(result.used).toBe(ipCount ?? count ?? 0);
    expect(result.limit).toBe(ipCount === null ? 50 : 1500);
    expect(result.remaining).toBe(ipCount === null ? Math.max(0, 50 - (count ?? 0)) : 0);
    expect(db.query('SELECT daily_count FROM usage WHERE device_id=?').get('synthetic-device')?.daily_count ?? 0).toBe(count ?? 0);
    expect(fetches).toBe(0);
  } finally { globalThis.fetch = original; db.close(); }
});
