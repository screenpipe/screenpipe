import { beforeEach, expect, mock, test } from 'bun:test';
let slots, index, scenario, calls;
// Deterministic state port: executes the exported hook, not React lifecycle.
mock.module('react', () => ({ useState: initial => { const i=index++; slots[i]=initial; return [initial, v => { slots[i]=typeof v==='function'?v(slots[i]):v; }]; }, useRef: v => ({ current:v }), useCallback: fn => fn }));
mock.module('gt-react', () => ({ useGT: () => (text, values) => text.replace('{value1, plural, one {# model} other {# models}}', String(values.value1)+' models') }));
const normalized = headers => Object.fromEntries(new Headers(headers).entries());
const expectedHeaders = () => ({ ...(scenario.key?{authorization:'Bearer '+scenario.key}:{}), ...normalized(scenario.headers) });
const authorized = headers => Object.entries(expectedHeaders()).every(([key,value]) => normalized(headers)[key]===value);
async function transport(url, init={}) {
  if (!String(url).startsWith('https://fixture.invalid')) throw Error('Unexpected external request');
  calls.push({url:String(url), init});
  if (String(url).endsWith('/v1/models')) return Response.json({ data:[{id:'synthetic-model'}] }, {status:scenario.reject || !authorized(init.headers)?401:200});
  if (String(url).endsWith('/v1/audio/transcriptions')) return Response.json({text:'synthetic'}, {status:scenario.txFailure?422:authorized(init.headers)?200:403});
  return Response.json({});
}
globalThis.fetch=transport;
mock.module('@/lib/http/tauri-fetch', () => ({ tauriFetchWithDeadline:transport }));
mock.module('@/lib/utils/tauri', () => ({ commands: { testOpenaiCompatibleTranscription:async(endpoint,key,model,headers) => {
  const merged={...(key?{Authorization:'Bearer '+key}:{}), ...(typeof headers==='string'?JSON.parse(headers):headers)};
  calls.push({native:true,endpoint,model,headers:merged});
  return scenario.txFailure || !authorized(merged)?{status:'error',error:'422 synthetic refusal'}:{status:'ok',data:'synthetic'};
} } }));
const { useTranscriptionDiagnostics } = await import('./use-transcription-diagnostics');
beforeEach(() => { slots=[];index=0;calls=[]; });
for (const value of [
  {name:'custom key without bearer',headers:{'X-Fixture-Key':'synthetic-only'}},
  {name:'custom Authorization overrides default bearer',key:'synthetic-default',headers:{Authorization:'Fixture synthetic-override','X-Fixture-Key':'synthetic-only'}},
  {name:'existing bearer',key:'synthetic-default'},
  {name:'public endpoint'},
  {name:'auth refusal skips transcription',key:'synthetic-default',reject:true},
  {name:'custom header auth refusal',headers:{'X-Fixture-Key':'synthetic-only'},reject:true},
  {name:'transcription refusal remains failure',key:'synthetic-default',txFailure:true},
]) test(value.name, async () => {
  scenario=value;
  const hook=useTranscriptionDiagnostics({settings:{openaiCompatibleEndpoint:'https://fixture.invalid',openaiCompatibleApiKey:value.key||'',openaiCompatibleModel:'synthetic-model',openaiCompatibleHeaders:value.headers}});
  await hook.runTranscriptionDiagnostics();
  expect(slots).toContain('done');
  const outcomes=slots.find(value=>value && typeof value==='object' && 'auth' in value && 'transcribe' in value);
  expect(outcomes).toBeDefined();
  const models=calls.filter(c=>c.url?.endsWith('/v1/models'));
  expect(models.length).toBeGreaterThan(0);
  // Refusal is a service outcome; the request still carries configured headers.
  expect(authorized(models.at(-1).init.headers)).toBe(true);
  const tx=calls.filter(c=>c.native || c.url?.endsWith('/v1/audio/transcriptions'));
  if(value.reject) { expect(outcomes.auth.status).toBe('fail');expect(outcomes.transcribe.status).toBe('skip');expect(tx).toHaveLength(0);return; }
  expect(outcomes.auth.status).toBe('pass');expect(tx).toHaveLength(1);
  const sent=tx[0];expect(authorized(sent.native?sent.headers:sent.init.headers)).toBe(true);
  if(sent.native) { expect(sent.endpoint).toBe('https://fixture.invalid');expect(sent.model).toBe('synthetic-model'); }
  else { expect(sent.init.method).toBe('POST');expect(sent.init.body instanceof FormData).toBe(true);expect(sent.init.body.get('model')).toBe('synthetic-model');expect(sent.init.body.get('file').size).toBeGreaterThan(44);expect(sent.init.body.get('file').type).toBe('audio/wav'); }
  expect(outcomes.transcribe.status).toBe(value.txFailure?'fail':'pass');
});
