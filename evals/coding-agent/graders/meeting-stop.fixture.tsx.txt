test('latest draft wins even if an older in-flight save completes later',async()=>{await mount();await edit('older');await tick();const first=p.requests[0];expect(first).toBeTruthy();await edit('latest');await tick();const completed=new Set<any>();if(p.requests[1]){await finish(p.requests[1]);completed.add(p.requests[1])}await finish(first);completed.add(first);for(const r of p.requests)if(!completed.has(r))await finish(r);expect(p.persisted).toBe('latest');expect((screen.getByRole('textbox',{name:'Synthetic note editor'}) as HTMLTextAreaElement).value).toBe('latest');});
test('Stop waits until the current draft is persisted',async()=>{await mount();await edit('latest');await stop();expect(p.stop).not.toHaveBeenCalled();expect(p.requests.length).toBeGreaterThan(0);await finish(p.requests[0]);expect(p.stop).toHaveBeenCalledWith('latest');});
test('failed save warns without blocking explicit Stop or pretending the note was saved',async()=>{await mount();await edit('latest');await stop();expect(p.requests.length).toBeGreaterThan(0);await finish(p.requests[0],false);expect(p.stop).toHaveBeenCalledTimes(1);expect(p.toast.mock.calls.some(([v])=>Boolean(v?.title||v?.description)) || screen.queryAllByRole('alert').some(el=>Boolean(el.textContent?.trim()))).toBe(true);expect(p.saved).not.toHaveBeenCalled();expect(screen.queryAllByText(/^saved\s*·/i)).toHaveLength(0);expect(p.persisted).toBe('initial');});
