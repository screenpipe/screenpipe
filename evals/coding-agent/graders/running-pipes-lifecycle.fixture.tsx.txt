// screenpipe — AI that knows everything you've seen, said, or heard
// https://screenpipe.com
import { beforeEach, afterEach, it, expect, vi } from 'vitest';
import { renderHook, act, cleanup } from '@testing-library/react';
import './lib/hooks/use-running-pipes';
const f=vi.hoisted(()=>({fetch:vi.fn(),mount:vi.fn(),listeners:new Set<any>()}));
vi.mock('@/lib/api',()=>({localFetch:f.fetch}));
vi.mock('@/lib/events/bus',()=>{const register=(fn:any)=>{f.listeners.add(fn);return ()=>f.listeners.delete(fn)};return {mountAgentEventBus:f.mount,registerDefault:register,registerObserver:register}});
const tick=async()=>{await act(async()=>{for(let i=0;i<8;i++)await Promise.resolve()})};
const row=(name:string,id=1,started='2026-01-01T01:00:00Z')=>({is_running:true,config:{name,config:{title:name}},recent_executions:[{id,status:'running',started_at:started}]});
const response=(data:any[])=>({ok:true,json:async()=>({data})});
const deferred=()=>{let resolve!:(v:any)=>void;const promise=new Promise<any>(r=>resolve=r);return {promise,resolve}};
let useRunningPipes:any;
beforeEach(async()=>{vi.resetModules();vi.useFakeTimers();vi.setSystemTime(new Date('2026-01-01T02:00:00Z'));f.listeners.clear();f.fetch.mockReset().mockResolvedValue(response([row('one')]));f.mount.mockReset().mockResolvedValue(undefined);globalThis.fetch=vi.fn(()=>{throw Error('network forbidden')}) as any;useRunningPipes=(await import('./lib/hooks/use-running-pipes')).useRunningPipes});
afterEach(()=>{cleanup();vi.clearAllTimers();vi.useRealTimers();vi.unstubAllGlobals()});
it('stops polling and event listening after the final consumer leaves',async()=>{const h=renderHook(()=>useRunningPipes());await tick();expect(h.result.current.map((x:any)=>x.pipeName)).toEqual(['one']);h.unmount();await tick();const before=f.fetch.mock.calls.length;await act(async()=>{await vi.advanceTimersByTimeAsync(90000)});expect(f.fetch).toHaveBeenCalledTimes(before);expect(f.listeners.size).toBe(0)});
it('preserves a shared tracker while another consumer remains',async()=>{const a=renderHook(()=>useRunningPipes());const b=renderHook(()=>useRunningPipes());await tick();expect(f.fetch).toHaveBeenCalledTimes(1);expect(f.listeners.size).toBe(1);a.unmount();f.fetch.mockResolvedValue(response([row('two')]));await act(async()=>{await vi.advanceTimersByTimeAsync(30000)});expect(b.result.current.map((x:any)=>x.pipeName)).toEqual(['two']);expect(f.listeners.size).toBe(1);b.unmount()});
it('does not start work when the initial fetch resolves after unmount',async()=>{const d=deferred();f.fetch.mockReturnValueOnce(d.promise);const h=renderHook(()=>useRunningPipes());h.unmount();d.resolve(response([row('late')]));await tick();await act(async()=>{await vi.advanceTimersByTimeAsync(90000)});expect(f.fetch).toHaveBeenCalledTimes(1);expect(f.listeners.size).toBe(0)});
it('starts a fresh fetch after all consumers unmount and remount',async()=>{const a=renderHook(()=>useRunningPipes());await tick();a.unmount();f.fetch.mockResolvedValue(response([row('new')]));const b=renderHook(()=>useRunningPipes());await tick();expect(b.result.current.map((x:any)=>x.pipeName)).toEqual(['new']);expect(f.listeners.size).toBe(1);b.unmount()});
it('ignores an old initial fetch after a new consumer has mounted',async()=>{const d=deferred();f.fetch.mockReturnValueOnce(d.promise);const a=renderHook(()=>useRunningPipes());a.unmount();f.fetch.mockResolvedValue(response([row('fresh')]));const b=renderHook(()=>useRunningPipes());await tick();d.resolve(response([row('stale')]));await tick();expect(b.result.current.map((x:any)=>x.pipeName)).toEqual(['fresh']);expect(f.listeners.size).toBe(1);b.unmount()});
it('late event-bus setup cannot install an abandoned subscription',async()=>{const d=deferred();f.mount.mockReturnValueOnce(d.promise);const a=renderHook(()=>useRunningPipes());await tick();a.unmount();d.resolve(undefined);await tick();expect(f.listeners.size).toBe(0);const before=f.fetch.mock.calls.length;await act(async()=>{await vi.advanceTimersByTimeAsync(60000)});expect(f.fetch).toHaveBeenCalledTimes(before)});
it('old event-bus completion preserves a newer active tracker',async()=>{const d=deferred();f.mount.mockReturnValueOnce(d.promise);const a=renderHook(()=>useRunningPipes());await tick();a.unmount();const b=renderHook(()=>useRunningPipes());await tick();d.resolve(undefined);await tick();expect(f.listeners.size).toBe(1);f.fetch.mockResolvedValue(response([row('newer')]));await act(async()=>{await vi.advanceTimersByTimeAsync(30000)});expect(b.result.current.map((x:any)=>x.pipeName)).toEqual(['newer']);b.unmount();expect(f.listeners.size).toBe(0)});
it('preserves pipe events and ignores unrelated events',async()=>{f.fetch.mockResolvedValue(response([]));const a=renderHook(()=>useRunningPipes());await tick();const send=(source:string,id:string,type:string)=>act(()=>{for(const fn of f.listeners)fn({source,sessionId:id,event:{type}})});send('pi','pipe:unrelated:8','message');expect(a.result.current).toEqual([]);send('pipe','not-a-pipe','message');expect(a.result.current).toEqual([]);send('pipe','pipe:live:9','message');expect(a.result.current.map((x:any)=>[x.pipeName,x.executionId])).toEqual([['live',9]]);send('pipe','pipe:live:9','pipe_done');expect(a.result.current).toEqual([]);a.unmount()});
it('preserves running-only filtering and stable start-time order',async()=>{f.fetch.mockResolvedValue(response([row('z',2,'2026-01-01T00:00:00Z'),row('a',1),{...row('stopped'),is_running:false}]));const a=renderHook(()=>useRunningPipes());await tick();expect(a.result.current.map((x:any)=>x.pipeName)).toEqual(['z','a']);a.unmount()});
it('recovers polling after a temporary HTTP failure',async()=>{f.fetch.mockResolvedValueOnce({ok:false});const a=renderHook(()=>useRunningPipes());await tick();expect(a.result.current).toEqual([]);await act(async()=>{await vi.advanceTimersByTimeAsync(30000)});expect(a.result.current.map((x:any)=>x.pipeName)).toEqual(['one']);a.unmount()});
