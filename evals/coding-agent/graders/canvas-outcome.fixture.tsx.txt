// Native commands store deep copies, allowing assertions about committed state.
let savedView: any;
let savedCanvas: any;
let writes: any[];
const copy = (v: any) => JSON.parse(JSON.stringify(v));
const note = { id: "safe-note", x: 90, y: 400, width: 200, height: 120, text: "Keep this annotation", color: "yellow" };
beforeAll(() => {
  Element.prototype.scrollIntoView ||= () => {};
  globalThis.PointerEvent ||= MouseEvent as typeof PointerEvent;
});
afterEach(cleanup);
beforeEach(() => {
  savedView = copy(populatedView);
  savedCanvas = { schema: "live-view-canvas.v1", viewId: savedView.id, revision: 7,
    mode: "canvas", viewport: { x: 24, y: 24, zoom: 1 },
    blocks: [{slotId: "focus-time", x: 64, y: 64, width: 440, height: 280}],
    notes: [copy(note)], arrows: [], strokes: [], updatedAt: "2026-07-27T17:00:00Z" };
  writes = [];
  mocks.listBrainViews.mockImplementation(async () => ({status: "ok", data: [savedView]}));
  mocks.loadBrainViewCanvas.mockImplementation(async () => ({status: "ok", data: savedCanvas}));
  mocks.saveBrainView.mockImplementation(async (request) => {
    request = copy(request);
    await new Promise(resolve => setTimeout(resolve, 0));
    savedView = {...savedView, ...copy(request), revision: savedView.revision + 1,
      slots: request.slots.map((slot: any) => ({...slot, value: null,
        feedback: {upCount: 0, downCount: 0, current: null}, itemActions: {items: []}}))};
    return {status: "ok", data: copy(savedView)};
  });
  mocks.saveBrainViewCanvas.mockImplementation(async (request) => {
    request = copy(request);
    writes.push(copy(request));
    await new Promise(resolve => setTimeout(resolve, 0));
    if (request.viewId !== savedView.id || request.blocks.some((b: any) => !savedView.slots.some((s: any) => s.id === b.slotId))) {
      return {status: "error", error: "canvas position does not belong to the saved view"};
    }
    savedCanvas = {...copy(request), schema: "live-view-canvas.v1", revision: savedCanvas.revision + 1, updatedAt: "2026-07-27T18:00:00Z"};
    return {status: "ok", data: savedCanvas};
  });
  mocks.generateLiveViewWithPi.mockResolvedValue({title: savedView.title, timeRange: savedView.timeRange,
    periodPolicy: savedView.periodPolicy, note: "Added a block", blocks: [
      {id: "focus-time", title: "Focus time", intent: "Calculate focused work time", component: "metric.v1", width: 6, pipeName: "daily-summary"},
      {id: "habit-signals", title: "Habit signals", intent: "Show recurring work habits", component: "list.v1", width: 6, pipeName: "daily-summary"}
    ]});
});
async function stage() {
  render(<BrainOverview />);
  await screen.findByLabelText(/Live View process canvas/i);
  fireEvent.change(await screen.findByPlaceholderText(/Ask AI to change this Live View/), {target: {value: "add habit signals"}});
  fireEvent.click(screen.getByRole("button", {name: /^update Live View$/i}));
  await screen.findByRole("button", {name: /^accept change to Habit signals$/i});
  expect(savedView.slots.map((s: any) => s.id)).toEqual(["focus-time"]);
}
async function zoom(requireSavedPositions = false) {
  fireEvent.click(await screen.findByRole("button", {name: /open canvas tools/i}));
  fireEvent.click(screen.getByRole("button", {name: /^reset zoom to 100%$/i}));
  fireEvent.click(screen.getByRole("button", {name: /^zoom in$/i}));
  if (requireSavedPositions) {
    await waitFor(() => expect(writes.length).toBeGreaterThan(0));
    expect(writes.every(w => w.blocks.every((b: any) => b.slotId === "focus-time")), "native writes must exclude unaccepted preview-only positions").toBe(true);
  }
  await waitFor(() => expect(savedCanvas.viewport.zoom).toBeGreaterThan(1));
}
function preserved(expectedNotes = [note]) {
  expect(savedCanvas.notes).toEqual(expectedNotes);
  expect(savedCanvas.blocks.map((b: any) => b.slotId)).toContain("focus-time");
  expect(savedCanvas.viewId).toBe(populatedView.id);
}
it("saves viewport edits during review without sending unaccepted positions", async () => {
  mocks.loadBrainViewCanvas.mockResolvedValue({status: "ok", data: null});
  savedCanvas = {...savedCanvas, revision: 0, notes: []};
  await stage();
  await zoom(true);
  expect(writes.length).toBeGreaterThan(0);
  expect(writes.every(w => w.blocks.every((b: any) => b.slotId === "focus-time"))).toBe(true);
  preserved([]);
  expect(mocks.toast.mock.calls.some(([v]) => v.title === "canvas changes were not saved")).toBe(false);
});
it("preserves normal canvas editing without a proposal", async () => {
  render(<BrainOverview />);
  await screen.findByRole("button", {name: /^edit Focus time with AI$/i});
  await zoom();
  preserved();
});
it("discarding a proposed block leaves the saved view and annotations intact", async () => {
  await stage();
  fireEvent.click(screen.getByRole("button", {name: /^reject all$/i}));
  await waitFor(() => expect(screen.queryByRole("button", {name: /^accept all$/i})).toBeNull());
  expect(savedView.slots.map((s: any) => s.id)).toEqual(["focus-time"]);
  expect(mocks.saveBrainView).not.toHaveBeenCalled();
  await zoom();
  preserved();
});
it("accepted proposals still persist the added block and allow later canvas edits", async () => {
  await stage();
  fireEvent.click(screen.getByRole("button", {name: /^accept all$/i}));
  await waitFor(() => expect(savedView.slots.map((s: any) => s.id)).toContain("habit-signals"));
  await waitFor(() => expect(screen.queryByRole("button", {name: /^accept all$/i})).toBeNull());
  await zoom();
  expect(savedCanvas.blocks.map((b: any) => b.slotId).sort()).toEqual(["focus-time", "habit-signals"]);
  preserved();
});

it("saves annotation edits while an added block is still only a proposal", async () => {
  await stage();
  const editor = await screen.findByRole("textbox", {name: /Canvas note/i});
  fireEvent.change(editor, {target: {value: "Updated safe annotation"}});
  fireEvent.blur(editor);
  await waitFor(() => expect(savedCanvas.notes[0].text).toBe("Updated safe annotation"));
  expect(savedView.slots.map((s: any) => s.id)).toEqual(["focus-time"]);
  expect(savedCanvas.blocks.map((b: any) => b.slotId)).toEqual(["focus-time"]);
  expect(savedCanvas.notes).toEqual([{...note, text: "Updated safe annotation"}]);
});
