import { act, render, screen } from "@testing-library/react";
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";

const mocks = vi.hoisted(() => ({
  localFetch: vi.fn(),
  spawnScreenpipe: vi.fn(),
  stopCapture: vi.fn(),
  startCapture: vi.fn(),
  getBootPhase: vi.fn(),
  handleNextSlide: vi.fn(),
  capture: vi.fn(),
  updateSettings: vi.fn(async () => undefined),
  settings: {
    aiPresets: [{}],
    user: null,
    disableScreenshots: false,
  },
}));

vi.mock("@/lib/api", () => ({ localFetch: mocks.localFetch }));
vi.mock("@/lib/utils/tauri", () => ({
  commands: {
    getAppIdentifier: vi.fn(async () => "com.screenpipe.app"),
    getBootPhase: mocks.getBootPhase,
    spawnScreenpipe: mocks.spawnScreenpipe,
    stopCapture: mocks.stopCapture,
    startCapture: mocks.startCapture,
  },
}));
vi.mock("@/lib/hooks/use-settings", () => ({
  useSettings: () => ({
    settings: mocks.settings,
    updateSettings: mocks.updateSettings,
  }),
  makeDefaultPresets: vi.fn(() => []),
}));
vi.mock("@/lib/utils/permission-flow", () => ({
  openPermissionSettingsWithFlow: vi.fn(),
}));
vi.mock("@tauri-apps/plugin-shell", () => ({ open: vi.fn() }));
vi.mock("@tauri-apps/plugin-opener", () => ({ revealItemInDir: vi.fn() }));
vi.mock("@tauri-apps/api/path", () => ({
  homeDir: vi.fn(async () => "/tmp"),
  join: vi.fn(async (...parts: string[]) => parts.join("/")),
}));
vi.mock("@tauri-apps/plugin-fs", () => ({ readTextFile: vi.fn() }));
vi.mock("@tauri-apps/api/app", () => ({
  getVersion: vi.fn(async () => "2.6.29"),
}));
vi.mock("@tauri-apps/plugin-os", () => ({
  platform: vi.fn(() => "macos"),
  version: vi.fn(() => "15.0"),
}));
vi.mock("posthog-js", () => ({ default: { capture: mocks.capture } }));
vi.mock("framer-motion", () => ({
  AnimatePresence: ({ children }: { children: React.ReactNode }) => children,
  motion: new Proxy(
    {},
    { get: (_target, element: string) => element },
  ),
}));
vi.mock("./particle-stream", () => ({
  ParticleStream: () => <div />,
  ProgressSteps: () => <div />,
}));

import EngineStartup from "./engine-startup";

const pendingBootPhase = {
  phase: "building_audio",
  message: "starting audio pipeline",
  error: null,
  sinceEpochSecs: 1,
};


const ok = { status: "ok", data: null };
const failed = { status: "error", error: "synthetic capture startup failure" };
async function elapse(ms: number) {
  for (let spent = 0; spent < ms; spent += 100) {
    await act(async () => { await vi.advanceTimersByTimeAsync(Math.min(100, ms - spent)); });
  }
}
function health(status = 503, body = { frame_status: "not_started", audio_status: "not_started" }) {
  mocks.localFetch.mockImplementation(async () => new Response(JSON.stringify(body), { status }));
}
beforeEach(() => {
  vi.useFakeTimers();
  vi.clearAllMocks();
  for (const value of Object.values(mocks)) { if (vi.isMockFunction(value)) value.mockReset(); }
  mocks.getBootPhase.mockResolvedValue(pendingBootPhase);
  mocks.spawnScreenpipe.mockResolvedValue(ok);
  mocks.stopCapture.mockResolvedValue(ok);
  mocks.startCapture.mockResolvedValue(ok);
  mocks.updateSettings.mockResolvedValue(undefined);
  mocks.settings.disableScreenshots = false;
  health();
  vi.spyOn(console, "error").mockImplementation(() => {});
});
afterEach(() => { vi.useRealTimers(); vi.restoreAllMocks(); });

describe("capture startup outcomes", () => {
  it("recovers from a returned startup error and advances once", async () => {
    mocks.startCapture.mockResolvedValueOnce(failed).mockResolvedValue(ok);
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2500);
    expect(mocks.handleNextSlide).toHaveBeenCalledTimes(1);
    expect(mocks.startCapture).toHaveBeenCalledTimes(2);
    expect(mocks.stopCapture).not.toHaveBeenCalled();
  });
  it("recovers from a thrown startup error and advances once", async () => {
    mocks.startCapture.mockRejectedValueOnce(new Error("synthetic temporary failure")).mockResolvedValue(ok);
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2500);
    expect(mocks.handleNextSlide).toHaveBeenCalledTimes(1);
    expect(mocks.startCapture).toHaveBeenCalledTimes(2);
    expect(mocks.stopCapture).not.toHaveBeenCalled();
  });
  it("reuses a successful session without stopping or respawning it", async () => {
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2500);
    expect(mocks.handleNextSlide).toHaveBeenCalledTimes(1);
    expect(mocks.startCapture).toHaveBeenCalledTimes(1);
    expect(mocks.stopCapture).not.toHaveBeenCalled();
    expect(mocks.spawnScreenpipe).not.toHaveBeenCalled();
  });
  it("waits for pending startup without overlapping native attempts", async () => {
    let finish: (value: typeof ok) => void = () => {};
    mocks.startCapture.mockImplementation(() => new Promise(resolve => { finish = resolve; }));
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2000);
    expect(mocks.startCapture).toHaveBeenCalledTimes(1);
    expect(mocks.handleNextSlide).not.toHaveBeenCalled();
    await act(async () => finish(ok));
    await elapse(1000);
    expect(mocks.handleNextSlide).toHaveBeenCalledTimes(1);
    expect(mocks.startCapture).toHaveBeenCalledTimes(1);
    expect(mocks.stopCapture).not.toHaveBeenCalled();
  });
  it("does not advance while capture keeps refusing startup", async () => {
    mocks.startCapture.mockResolvedValue(failed);
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2500);
    expect(mocks.startCapture).toHaveBeenCalled();
    expect(mocks.handleNextSlide).not.toHaveBeenCalled();
    expect(screen.getByText(failed.error)).toBeTruthy();
    expect(mocks.stopCapture).not.toHaveBeenCalled();
  });
  it("accepts initialized capture while media is intentionally idle", async () => {
    mocks.settings.disableScreenshots = true;
    health(503, { frame_status: "stale", audio_status: "not_started" });
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2500);
    expect(mocks.handleNextSlide).toHaveBeenCalledTimes(1);
    expect(mocks.startCapture).toHaveBeenCalledTimes(1);
  });
  it("uses native capture readiness when local health requires authentication", async () => {
    mocks.localFetch.mockImplementation(async () => new Response("{}", { status: 401 }));
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2500);
    expect(mocks.handleNextSlide).toHaveBeenCalledTimes(1);
    expect(mocks.startCapture).toHaveBeenCalledTimes(1);
    expect(mocks.spawnScreenpipe).not.toHaveBeenCalled();
  });
  it("does not claim readiness from an unrelated health payload", async () => {
    mocks.localFetch.mockImplementation(async () => new Response("{}", { status: 200 }));
    render(<EngineStartup handleNextSlide={mocks.handleNextSlide} />);
    await elapse(2500);
    expect(mocks.handleNextSlide).not.toHaveBeenCalled();
    expect(mocks.startCapture).not.toHaveBeenCalled();
  });
});
